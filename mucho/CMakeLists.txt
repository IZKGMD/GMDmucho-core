cmake_minimum_required(VERSION 3.25)
project(MuchoRuntimeTests LANGUAGES CXX)
find_package(Python3 3.12 REQUIRED COMPONENTS Interpreter Development.Embed)
find_package(Threads REQUIRED)
add_library(mucho-runtime STATIC native/Runtime.cpp)
target_compile_features(mucho-runtime PUBLIC cxx_std_20)
target_include_directories(mucho-runtime PUBLIC native)
target_link_libraries(mucho-runtime PRIVATE Python3::Python Threads::Threads)
add_executable(mucho-runtime-tests tests/runtime.cpp)
target_link_libraries(mucho-runtime-tests PRIVATE mucho-runtime)
enable_testing()
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import sys;print(sys.prefix)" OUTPUT_VARIABLE MUCHO_TEST_PYTHON_HOME OUTPUT_STRIP_TRAILING_WHITESPACE)
add_test(NAME embedded-runtime COMMAND mucho-runtime-tests "${CMAKE_CURRENT_SOURCE_DIR}" "${MUCHO_TEST_PYTHON_HOME}")
add_test(NAME python-sdk COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_sdk.py")
